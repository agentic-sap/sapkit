```abap
*&---------------------------------------------------------------------*
*& Report ZSAPKIT_DEMO_PROG
*&---------------------------------------------------------------------*
*& Demo report: company code / business place selection with validation
*&---------------------------------------------------------------------*
REPORT zsapkit_demo_prog.

*----------------------------------------------------------------------*
* Data declarations
*----------------------------------------------------------------------*
TABLES: t001.

DATA: gv_butxt TYPE t001-butxt,
      gv_bupla TYPE j_1bbranch-branch,
      gv_name  TYPE j_1bbranch-name.

*----------------------------------------------------------------------*
* Selection screen
*----------------------------------------------------------------------*
SELECTION-SCREEN BEGIN OF BLOCK b01 WITH FRAME TITLE TEXT-b01.
  PARAMETERS: pa_bukrs TYPE t001-bukrs       OBLIGATORY,
              pa_bupla TYPE j_1bbranch-branch.
SELECTION-SCREEN END OF BLOCK b01.

*----------------------------------------------------------------------*
* INITIALIZATION
*----------------------------------------------------------------------*
INITIALIZATION.
  GET PARAMETER ID 'BUK' FIELD pa_bukrs.
  IF pa_bukrs IS INITIAL.
    pa_bukrs = '1000'.
  ENDIF.

*----------------------------------------------------------------------*
* AT SELECTION-SCREEN ON pa_bukrs
*----------------------------------------------------------------------*
AT SELECTION-SCREEN ON pa_bukrs.
  SELECT SINGLE butxt
    FROM t001
    INTO @gv_butxt
    WHERE bukrs = @pa_bukrs.
  IF sy-subrc <> 0.
    MESSAGE e001(00) WITH 'Company code' pa_bukrs 'does not exist'.
  ENDIF.

  AUTHORITY-CHECK OBJECT 'F_BKPF_BUK'
    ID 'BUKRS' FIELD pa_bukrs
    ID 'ACTVT' FIELD '03'.
  IF sy-subrc <> 0.
    MESSAGE e001(00) WITH 'No authorization for company code' pa_bukrs.
  ENDIF.

*----------------------------------------------------------------------*
* AT SELECTION-SCREEN ON pa_bupla
*----------------------------------------------------------------------*
AT SELECTION-SCREEN ON pa_bupla.
  CHECK pa_bupla IS NOT INITIAL.

  " Business place must exist for the entered company code
  SELECT SINGLE branch, name
    FROM j_1bbranch
    INTO (@gv_bupla, @gv_name)
    WHERE bukrs  = @pa_bukrs
      AND branch = @pa_bupla.
  IF sy-subrc <> 0.
    MESSAGE e001(00) WITH 'Business place' pa_bupla
                          'not defined for company code' pa_bukrs.
  ENDIF.

*----------------------------------------------------------------------*
* START-OF-SELECTION
*----------------------------------------------------------------------*
START-OF-SELECTION.
  SELECT SINGLE butxt
    FROM t001
    INTO @gv_butxt
    WHERE bukrs = @pa_bukrs.

  WRITE: / 'Company code :', pa_bukrs, gv_butxt.

  IF pa_bupla IS NOT INITIAL.
    SELECT SINGLE name
      FROM j_1bbranch
      INTO @gv_name
      WHERE bukrs  = @pa_bukrs
        AND branch = @pa_bupla.
    WRITE: / 'Business place:', pa_bupla, gv_name.
  ELSE.
    WRITE: / 'Business place:', '(all)'.
  ENDIF.
```
